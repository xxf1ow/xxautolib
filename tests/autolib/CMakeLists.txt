cmake_minimum_required(VERSION 3.20)
project(autolib_smoke LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT MODULES)
    set(MODULES queue lz4 flann)
endif()
set(module_names "")
set(smoke_modules queue lz4 flann eigen freetype harfbuzz libusb nlohmann opencv librealsense
    asio clipper2 cpp-httplib expected gtest libzmq nativefiledialog-extended oneTBB portaudio pybind11 sdl3 spdlog taskflow yaml-cpp zlib
    boost websocketpp imgui pcl visp)
foreach(ref IN LISTS MODULES)
    if(NOT ref MATCHES "^([A-Za-z0-9][A-Za-z0-9_.+\\-]*)(@([A-Za-z0-9][A-Za-z0-9_.+\\-]*))?$")
        message(FATAL_ERROR "invalid consumer module ${ref}")
    endif()
    set(name "${CMAKE_MATCH_1}")
    if(name STREQUAL "onnxruntime")
        message(STATUS "ONNX Runtime upstream precompiled package: API smoke exempt; installation and import checks remain enabled")
        continue()
    endif()
    if(NOT name IN_LIST smoke_modules)
        message(FATAL_ERROR "No API smoke implementation for ${name}")
    endif()
    if(name STREQUAL "taskflow")
        message(STATUS "Taskflow C++20 API smoke skipped; installation and import checks remain enabled")
        continue()
    endif()
    list(APPEND module_names "${name}")
endforeach()
list(REMOVE_DUPLICATES module_names)
add_executable(autolib_smoke smoke.cpp)
if(MSVC)
    target_compile_options(autolib_smoke PRIVATE /Zc:__cplusplus)
endif()

include("${AUTOLIB_ENTRY}")
if(NOT ROOT_PACKAGES)
    set(ROOT_PACKAGES ${MODULES})
endif()
autolib_import(TARGET autolib_smoke PRIVATE ${ROOT_PACKAGES})
# 包来源检查只记录包导入时创建的目标，不包含测试随后发现的 Python 等环境依赖。
get_property(package_targets DIRECTORY PROPERTY IMPORTED_TARGETS)
if(ROOT_PACKAGES STREQUAL "flann;queue" AND NOT AUTOLIB_PREPARED_PACKAGES STREQUAL "lz4;flann;queue")
    message(FATAL_ERROR "Root-only consumption did not prepare lz4;flann;queue: ${AUTOLIB_PREPARED_PACKAGES}")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/prepared-packages.txt" "${AUTOLIB_PREPARED_PACKAGES}")
foreach(name IN LISTS module_names)
    string(MAKE_C_IDENTIFIER "${name}" macro_name)
    string(TOUPPER "${macro_name}" upper_name)
    target_compile_definitions(autolib_smoke PRIVATE AUTOLIB_TEST_${upper_name}=1)
    if(NOT AUTOLIB_PACKAGE_VERSION_${name})
        message(FATAL_ERROR "Requested smoke module was not prepared: ${name}")
    endif()
    if(name MATCHES "^(flann|opencv|librealsense)$")
        target_compile_definitions(autolib_smoke PRIVATE "AUTOLIB_EXPECT_${upper_name}_ROOT=\"${AUTOLIB_PACKAGE_DIR_${name}}\"")
    endif()
endforeach()
if(WIN32)
    if(opencv IN_LIST module_names)
        target_compile_definitions(autolib_smoke PRIVATE "AUTOLIB_OPENCV_DLL=L\"$<TARGET_FILE_NAME:opencv_core>\"")
    endif()
    if(librealsense IN_LIST module_names)
        target_compile_definitions(autolib_smoke PRIVATE "AUTOLIB_LIBREALSENSE_DLL=L\"$<TARGET_FILE_NAME:realsense2::realsense2>\"")
    endif()
endif()
if(eigen IN_LIST module_names)
    target_compile_definitions(autolib_smoke PRIVATE "AUTOLIB_EXPECT_EIGEN_VERSION=\"${AUTOLIB_PACKAGE_VERSION_eigen}\"")
endif()
# WebSocket++ 单包使用内置 iostream transport；显式组合 Asio 时验证该后端。
if(websocketpp IN_LIST module_names AND asio IN_LIST module_names)
    target_compile_definitions(autolib_smoke PRIVATE ASIO_STANDALONE AUTOLIB_TEST_WEBSOCKETPP_ASIO)
endif()
if(flann IN_LIST module_names OR opencv IN_LIST module_names OR librealsense IN_LIST module_names)
    target_link_libraries(autolib_smoke PRIVATE ${CMAKE_DL_LIBS})
endif()
add_test(NAME local_binary_smoke COMMAND autolib_smoke)
if(pybind11 IN_LIST module_names)
    find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
    Python3_add_library(autolib_binding MODULE WITH_SOABI smoke.cpp)
    target_compile_definitions(autolib_binding PRIVATE AUTOLIB_PYBIND_MODULE=1)
    target_link_libraries(autolib_binding PRIVATE autolib::pybind11)
    add_test(NAME python_binding_smoke COMMAND "${Python3_EXECUTABLE}" -c
        "import sys; sys.path.insert(0, r'$<TARGET_FILE_DIR:autolib_binding>'); import autolib_binding; assert autolib_binding.add(20, 22) == 42")
endif()

get_filename_component(cache_root "${AUTOLIB_ENTRY}" DIRECTORY)
set(state "set(AUTOLIB_CACHE_ROOT [==[${cache_root}]==])\n")
foreach(name IN LISTS AUTOLIB_PREPARED_PACKAGES)
    string(APPEND state "set(AUTOLIB_PACKAGE_DIR_${name} [==[${AUTOLIB_PACKAGE_DIR_${name}}]==])\nset(AUTOLIB_PACKAGE_VERSION_${name} [==[${AUTOLIB_PACKAGE_VERSION_${name}}]==])\n")
endforeach()
if(WIN32)
    set(runtime_dirs "")
    foreach(name IN LISTS AUTOLIB_PREPARED_PACKAGES)
        file(GLOB_RECURSE runtime_files LIST_DIRECTORIES false "${AUTOLIB_PACKAGE_DIR_${name}}/*.dll")
        foreach(runtime_file IN LISTS runtime_files)
            get_filename_component(runtime_dir "${runtime_file}" DIRECTORY)
            list(APPEND runtime_dirs "${runtime_dir}")
        endforeach()
    endforeach()
    list(REMOVE_DUPLICATES runtime_dirs)
    string(APPEND state "set(AUTOLIB_RUNTIME_DIRS [==[${runtime_dirs}]==])\n")
endif()
if(flann IN_LIST module_names)
    if(WIN32)
        set(runtime_dir "${AUTOLIB_PACKAGE_DIR_flann}/bin")
    else()
        set(runtime_dir "${AUTOLIB_PACKAGE_DIR_flann}/lib")
    endif()
    string(APPEND state "set(AUTOLIB_FLANN_RUNTIME_DIR [==[${runtime_dir}]==])\n")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/prepared-state.cmake" "${state}")
set(inputs "")
set(link_inputs "")
foreach(target IN LISTS package_targets)
    get_target_property(kind "${target}" TYPE)
    if(NOT kind STREQUAL "INTERFACE_LIBRARY" AND NOT kind STREQUAL "EXECUTABLE")
        list(APPEND link_inputs "$<TARGET_FILE:${target}>")
        if(kind STREQUAL "SHARED_LIBRARY")
            list(APPEND link_inputs "$<TARGET_LINKER_FILE:${target}>")
        endif()
    endif()
    foreach(property IN ITEMS IMPORTED_LOCATION_RELEASE IMPORTED_IMPLIB_RELEASE INTERFACE_INCLUDE_DIRECTORIES INTERFACE_LINK_LIBRARIES)
        get_target_property(value "${target}" "${property}")
        if(value)
            string(APPEND inputs "${target}\t${property}\t${value}\n")
        endif()
    endforeach()
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/imported-properties.txt" "${inputs}")
string(JOIN "\n" link_input_text ${link_inputs})
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/link-inputs-$<CONFIG>.txt" CONTENT "${link_input_text}\n")
