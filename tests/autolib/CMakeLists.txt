cmake_minimum_required(VERSION 3.20)
project(autolib_smoke LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(AUTOLIB_SUPPORTED_MODULES queue lz4 flann)
if(NOT MODULES)
    set(MODULES ${AUTOLIB_SUPPORTED_MODULES})
endif()
foreach(name IN LISTS MODULES)
    if(NOT name IN_LIST AUTOLIB_SUPPORTED_MODULES)
        message(FATAL_ERROR "unsupported consumer module ${name}")
    endif()
endforeach()
set(selected_modules "")
foreach(name IN LISTS AUTOLIB_SUPPORTED_MODULES)
    if(name IN_LIST MODULES)
        list(APPEND selected_modules "${name}")
    endif()
endforeach()
set(MODULES "${selected_modules}")
add_executable(autolib_smoke smoke.cpp)

if(PREPARED_ROOT)
    foreach(name IN LISTS MODULES)
        set(AUTOLIB_PACKAGE_DIR_${name} "${PREPARED_ROOT}/${name}")
        include("${PREPARED_ROOT}/${name}/autolib-import.cmake")
        target_link_libraries(autolib_smoke PRIVATE autolib::${name})
    endforeach()
else()
    set(AUTOLIB_LOCAL_ROOT "${RELEASE_ROOT}")
    set(AUTOLIB_REMOTE_ROOT "")
    include("${AUTOLIB_ENTRY}")
    if(NOT ROOT_PACKAGES)
        set(ROOT_PACKAGES ${MODULES})
    endif()
    autolib_import(TARGET autolib_smoke PRIVATE ${ROOT_PACKAGES})
endif()
if(NOT PREPARED_ROOT AND ROOT_PACKAGES STREQUAL "flann;queue" AND NOT AUTOLIB_PREPARED_PACKAGES STREQUAL "lz4;flann;queue")
    message(FATAL_ERROR "Root-only consumption did not prepare lz4;flann;queue: ${AUTOLIB_PREPARED_PACKAGES}")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/prepared-packages.txt" "${AUTOLIB_PREPARED_PACKAGES}")
foreach(name IN LISTS MODULES)
    string(TOUPPER "${name}" upper_name)
    target_compile_definitions(autolib_smoke PRIVATE AUTOLIB_TEST_${upper_name}=1)
endforeach()
if(flann IN_LIST MODULES)
    target_link_libraries(autolib_smoke PRIVATE ${CMAKE_DL_LIBS})
endif()
add_test(NAME local_binary_smoke COMMAND autolib_smoke)

list(GET MODULES 0 first_module)
file(WRITE "${CMAKE_BINARY_DIR}/prepared-root.txt" "${AUTOLIB_PACKAGE_DIR_${first_module}}/..")
set(inputs "")
set(targets "")
set(link_inputs "")
if(lz4 IN_LIST MODULES)
    list(APPEND targets LZ4::lz4_static)
    list(APPEND link_inputs "$<TARGET_FILE:LZ4::lz4_static>")
endif()
if(flann IN_LIST MODULES)
    list(APPEND targets flann::flann_cpp flann::flann)
    list(APPEND link_inputs "$<TARGET_LINKER_FILE:flann::flann>" "$<TARGET_FILE:flann::flann>" "$<TARGET_FILE:flann::flann_cpp>")
endif()
foreach(target IN LISTS targets)
    foreach(property IN ITEMS IMPORTED_LOCATION_RELEASE IMPORTED_IMPLIB_RELEASE INTERFACE_INCLUDE_DIRECTORIES INTERFACE_LINK_LIBRARIES)
        get_target_property(value "${target}" "${property}")
        if(value)
            string(APPEND inputs "${target}\t${property}\t${value}\n")
        endif()
    endforeach()
endforeach()
if(INSTALL_ROOT)
    string(FIND "${inputs}" "${INSTALL_ROOT}" producer_position)
    if(NOT producer_position EQUAL -1)
        message(FATAL_ERROR "Imported targets retain producer paths:\n${inputs}")
    endif()
endif()
file(WRITE "${CMAKE_BINARY_DIR}/imported-properties.txt" "${inputs}")
string(JOIN "\n" link_input_text ${link_inputs})
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/link-inputs-$<CONFIG>.txt" CONTENT "${link_input_text}\n")
