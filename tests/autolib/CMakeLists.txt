cmake_minimum_required(VERSION 3.20)
project(autolib_smoke LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(PREPARED_ROOT)
    foreach(name IN ITEMS queue lz4 flann)
        set(AUTOLIB_PACKAGE_DIR_${name} "${PREPARED_ROOT}/${name}")
    endforeach()
    include("${PREPARED_ROOT}/flann/autolib-import.cmake")
    include("${PREPARED_ROOT}/queue/autolib-import.cmake")
    include("${PREPARED_ROOT}/lz4/autolib-import.cmake")
else()
    include("${AUTOLIB_ENTRY}")
    autolib_import(URI "${RELEASE_ROOT}" WORK_ROOT "${WORK_ROOT}"
        TARGET_OS "${TARGET_OS}" TARGET_ARCH "${TARGET_ARCH}" PACKAGES flann queue lz4 flann)
endif()
add_executable(autolib_smoke smoke.cpp)
target_link_libraries(autolib_smoke PRIVATE AutoLib::queue AutoLib::lz4 AutoLib::flann flann::flann ${CMAKE_DL_LIBS})
add_test(NAME local_binary_smoke COMMAND autolib_smoke)

file(WRITE "${CMAKE_BINARY_DIR}/prepared-root.txt" "${AUTOLIB_PACKAGE_DIR_flann}/..")
set(inputs "")
foreach(target IN ITEMS LZ4::lz4_static flann::flann_cpp flann::flann)
    foreach(property IN ITEMS IMPORTED_LOCATION_RELEASE IMPORTED_IMPLIB_RELEASE INTERFACE_INCLUDE_DIRECTORIES INTERFACE_LINK_LIBRARIES)
        get_target_property(value "${target}" "${property}")
        if(value)
            string(APPEND inputs "${target}\t${property}\t${value}\n")
        endif()
    endforeach()
endforeach()
if(INSTALL_ROOT)
    string(FIND "${inputs}" "${INSTALL_ROOT}" producer_position)
    if(NOT producer_position EQUAL -1)
        message(FATAL_ERROR "Imported targets retain producer paths:\n${inputs}")
    endif()
endif()
file(WRITE "${CMAKE_BINARY_DIR}/imported-properties.txt" "${inputs}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/link-inputs-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:LZ4::lz4_static>\n$<TARGET_LINKER_FILE:flann::flann>\n$<TARGET_FILE:flann::flann>\n$<TARGET_FILE:flann::flann_cpp>\n")
