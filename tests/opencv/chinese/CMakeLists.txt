cmake_minimum_required(VERSION 3.20)
project(opencv_chinese_check LANGUAGES CXX)
enable_testing()

foreach(input IN ITEMS AUTOLIB_ENTRY CHINESE_FONT)
    if(NOT IS_ABSOLUTE "${${input}}")
        message(FATAL_ERROR "${input} must be an absolute path")
    endif()
endforeach()

add_executable(opencv_chinese smoke.cpp)
target_compile_features(opencv_chinese PRIVATE cxx_std_17)
include("${AUTOLIB_ENTRY}")
autolib_import(TARGET opencv_chinese PRIVATE opencv@4.14.0)
add_test(NAME opencv_chinese COMMAND opencv_chinese "${CHINESE_FONT}" "${CMAKE_BINARY_DIR}/chinese.pgm")

set(runtime_dirs "")
foreach(name IN LISTS AUTOLIB_PREPARED_PACKAGES)
    if(WIN32)
        file(GLOB_RECURSE libraries LIST_DIRECTORIES false "${AUTOLIB_PACKAGE_DIR_${name}}/*.dll")
        foreach(library IN LISTS libraries)
            get_filename_component(directory "${library}" DIRECTORY)
            list(APPEND runtime_dirs "${directory}")
        endforeach()
    endif()
endforeach()
list(REMOVE_DUPLICATES runtime_dirs)
file(WRITE "${CMAKE_BINARY_DIR}/runtime-dirs.txt" "${runtime_dirs}")
