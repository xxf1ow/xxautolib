cmake_minimum_required(VERSION 3.20)
project(opencv_install_check CXX)

if(NOT IS_ABSOLUTE "${OPENCV_INSTALL_PREFIX}")
    message(FATAL_ERROR "OPENCV_INSTALL_PREFIX must be an absolute installation path")
endif()
if(WIN32)
    set(config_dir "${OPENCV_INSTALL_PREFIX}")
else()
    set(config_dir "${OPENCV_INSTALL_PREFIX}/lib/cmake/opencv4")
endif()
if(NOT EXISTS "${config_dir}/OpenCVConfig.cmake")
    message(FATAL_ERROR "Missing installed OpenCV config: ${config_dir}/OpenCVConfig.cmake")
endif()
set(OpenCV_DIR "${config_dir}")
find_package(OpenCV 4.14.0 EXACT REQUIRED CONFIG PATHS "${config_dir}" NO_DEFAULT_PATH)

add_executable(opencv_smoke smoke.cpp)
target_compile_features(opencv_smoke PRIVATE cxx_std_17)
target_include_directories(opencv_smoke PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(opencv_smoke PRIVATE ${OpenCV_LIBS})
enable_testing()
add_test(NAME opencv_install COMMAND opencv_smoke)
