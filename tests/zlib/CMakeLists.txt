cmake_minimum_required(VERSION 3.20)
project(zlib_install_check C)

if(NOT IS_ABSOLUTE "${ZLIB_INSTALL_PREFIX}")
    message(FATAL_ERROR "ZLIB_INSTALL_PREFIX must be an absolute installation path")
endif()
foreach(header zlib.h zconf.h)
    if(NOT EXISTS "${ZLIB_INSTALL_PREFIX}/include/${header}")
        message(FATAL_ERROR "Missing installed zlib header: ${ZLIB_INSTALL_PREFIX}/include/${header}")
    endif()
endforeach()
if(MSVC)
    set(static_name zlibstatic.lib)
else()
    set(static_name libz.a)
endif()
set(static_library "${ZLIB_INSTALL_PREFIX}/lib/${static_name}")
if(NOT EXISTS "${static_library}")
    message(FATAL_ERROR "Missing installed zlib static library: ${static_library}")
endif()

add_executable(zlib_smoke smoke.c)
target_include_directories(zlib_smoke PRIVATE "${ZLIB_INSTALL_PREFIX}/include")
target_link_libraries(zlib_smoke PRIVATE "${static_library}")
enable_testing()
add_test(NAME zlib_install COMMAND zlib_smoke)
