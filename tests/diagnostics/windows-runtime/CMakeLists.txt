cmake_minimum_required(VERSION 3.20)
project(windows_runtime_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(runtime_probe SHARED library.cpp)
set_target_properties(runtime_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package")
add_executable(consumer consumer.cpp)
target_link_libraries(consumer PRIVATE runtime_probe)
target_compile_definitions(consumer PRIVATE PROBE_ROOT=L"$<TARGET_FILE_DIR:runtime_probe>" PROBE_DLL=L"$<TARGET_FILE_NAME:runtime_probe>" PROBE_CONSUMER_ROOT=L"$<TARGET_FILE_DIR:consumer>")
enable_testing()
add_test(NAME import_thunk_diagnosis COMMAND "${CMAKE_COMMAND}" -E env --modify "PATH=path_list_prepend:$<TARGET_FILE_DIR:runtime_probe>" -- $<TARGET_FILE:consumer>)
